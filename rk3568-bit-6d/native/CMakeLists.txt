cmake_minimum_required(VERSION 3.16)
project(pose6d_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(OpenCV 4.5 REQUIRED COMPONENTS core imgproc calib3d aruco)
find_package(Threads REQUIRED)
find_path(RGA_INCLUDE_DIR im2d.h PATH_SUFFIXES rga REQUIRED)
find_library(RGA_LIBRARY NAMES rga REQUIRED)
get_filename_component(RGA_LIBRARY_DIR "${RGA_LIBRARY}" DIRECTORY)
add_executable(pose6d src/main.cpp)
target_include_directories(pose6d PRIVATE ${RGA_INCLUDE_DIR}
  ../src/monocular_6d_pose/include)
target_link_libraries(pose6d PRIVATE ${OpenCV_LIBS} ${RGA_LIBRARY} Threads::Threads dl)
target_compile_options(pose6d PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(pose6d PRIVATE -static-libstdc++ -static-libgcc)
target_link_options(pose6d PRIVATE "-Wl,-rpath-link,${RGA_LIBRARY_DIR}")
set_target_properties(pose6d PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
install(TARGETS pose6d DESTINATION .)
install(PROGRAMS run.sh DESTINATION .)
install(FILES analyze_logs.py DESTINATION .)
install(FILES README.md DEPENDENCIES.txt BOARD_VALIDATION.md DESTINATION .)
