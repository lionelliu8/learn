cmake_minimum_required(VERSION 3.8)
project(monocular_6d_pose)
set(CMAKE_CXX_STANDARD 14)
option(ENABLE_RGA "Build Rockchip librga im2d resize support" OFF)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core calib3d imgproc aruco)

add_executable(pose_node src/pose_node.cpp)
target_include_directories(pose_node PRIVATE include)
target_link_libraries(pose_node ${OpenCV_LIBS})
if(ENABLE_RGA)
  find_path(RGA_INCLUDE_DIR im2d.h PATH_SUFFIXES rga)
  find_library(RGA_LIBRARY NAMES rga)
  if(NOT RGA_INCLUDE_DIR OR NOT RGA_LIBRARY)
    message(FATAL_ERROR "Install board-matched librga headers/library or set RGA_INCLUDE_DIR and RGA_LIBRARY")
  endif()
  target_include_directories(pose_node PRIVATE ${RGA_INCLUDE_DIR})
  target_link_libraries(pose_node ${RGA_LIBRARY})
  target_compile_definitions(pose_node PRIVATE USE_RGA=1)
endif()
ament_target_dependencies(pose_node rclcpp sensor_msgs geometry_msgs std_msgs cv_bridge tf2 tf2_ros tf2_geometry_msgs)

add_executable(synthetic_camera_node src/synthetic_camera_node.cpp)
target_link_libraries(synthetic_camera_node ${OpenCV_LIBS})
ament_target_dependencies(synthetic_camera_node rclcpp sensor_msgs cv_bridge)

install(TARGETS pose_node synthetic_camera_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_pose_solver test/test_pose_solver.cpp)
  target_include_directories(test_pose_solver PRIVATE include)
  target_link_libraries(test_pose_solver ${OpenCV_LIBS})
endif()

ament_package()
